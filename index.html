local function isAreaOccupied(holder, col, row, w, h, ignoreBtn)
	for _, child in ipairs(holder:GetChildren()) do
		if child:IsA("ImageButton") and child ~= ignoreBtn then
			local cx = child:GetAttribute("Col")
			local cy = child:GetAttribute("Row")
			local cw, ch = itemDims(child.Name)
			if cx and cy then
				if col < cx + cw and col + w > cx and row < cy + ch and row + h > cy then
					return true
				end
			end
		end
	end
	return false
end

local function isAreaWithinValidSlots(col, row, w, h)
	for r = row, row + h - 1 do
		for c = col, col + w - 1 do
			if not isSlotIndexValid(c, r) then
				return false
			end
		end
	end
	return true
end

local function findFreeSlot(holder, w, h)
	local rows = holder == InvHolder and getTotalRows() or 6
	local cols = GRID_W
	for y = 1, rows - h + 1 do
		for x = 1, cols - w + 1 do
			local valid = true
			if holder == InvHolder then
				valid = isAreaWithinValidSlots(x, y, w, h)
			end
			if valid and not isAreaOccupied(holder, x, y, w, h) then
				return x, y
			end
		end
	end
	return nil, nil
end

local function getGhostCoord(holder, w, h)
	if cellSize == 0 then return nil, nil end
	local m = mousePos()
	local pos = holder.AbsolutePosition
	local cs = cellSize + PADDING
	local rows = holder == InvHolder and getTotalRows() or 6
	local col = math.floor((m.X - pos.X - PADDING) / cs - (w - 1) / 2) + 1
	local row = math.floor((m.Y - pos.Y - PADDING) / cs - (h - 1) / 2) + 1
	col = math.clamp(col, 1, GRID_W - w + 1)
	row = math.clamp(row, 1, rows - h + 1)
	return col, row
end

local function findRemote(containerObj, className, name)
	if not containerObj then return nil end
	local direct = containerObj:FindFirstChild(name)
	if direct and direct.ClassName == className then return direct end
	for _, child in ipairs(containerObj:GetDescendants()) do
		if child.Name == name and child.ClassName == className then return child end
	end
	return nil
end

local function resolveContainer()
	if container then return container end
	local sutfFolder = ReplicatedStorage:FindFirstChild("sutf")
	return sutfFolder or moduleScript
end

local function getRemoteEvent()
	if remoteEvent then return remoteEvent end
	local root = resolveContainer()
	remoteEvent = findRemote(root, "RemoteEvent", "RemoteEvent")
	if not remoteEvent and root ~= moduleScript then
		remoteEvent = findRemote(moduleScript, "RemoteEvent", "RemoteEvent")
	end
	if not remoteEvent then error("re not found.") end
	return remoteEvent
end

local function getRemoteFunction()
	if remoteFunction then return remoteFunction end
	local root = resolveContainer()
	remoteFunction = findRemote(root, "RemoteFunction", "RemoteFunction")
	if not remoteFunction and root ~= moduleScript then
		remoteFunction = findRemote(moduleScript, "RemoteFunction", "RemoteFunction")
	end
	return remoteFunction
end

local module = {}

function module.getUUID() return currentUUID end

function module.invoke(action, data)
	return getRemoteFunction():InvokeServer(action, data)
end

function module.fire(action, data)
	getRemoteEvent():FireServer(action, data)
end

local function setGuiTreeTransparency(root, transparency)
	for _, object in ipairs(collectGuiObjects(root)) do
		if object:IsA("Frame") then
			object.BackgroundTransparency = transparency
		elseif object:IsA("TextLabel") or object:IsA("TextButton") then
			object.TextTransparency = transparency
			object.TextStrokeTransparency = 1
		elseif object:IsA("ImageLabel") or object:IsA("ImageButton") then
			object.ImageTransparency = transparency
		end
	end
end

local function tweenGuiTree(root, transparency, duration)
	local tweens = {}
	for _, object in ipairs(collectGuiObjects(root)) do
		if object:IsA("Frame") then
			tweens[#tweens + 1] = TweenService:Create(object, TweenInfo.new(duration, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), { BackgroundTransparency = transparency })
		elseif object:IsA("TextLabel") or object:IsA("TextButton") then
			tweens[#tweens + 1] = TweenService:Create(object, TweenInfo.new(duration, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), { TextTransparency = transparency, TextStrokeTransparency = 1 })
		elseif object:IsA("ImageLabel") or object:IsA("ImageButton") then
			tweens[#tweens + 1] = TweenService:Create(object, TweenInfo.new(duration, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), { ImageTransparency = transparency })
		end
	end
	for _, tween in ipairs(tweens) do
		tween:Play()
	end
	if duration > 0 then
		task.wait(duration)
	end
end

local function clampToWater(worldPosition)
	local water = shared.WaterPart
	local ring = shared.DropRing
	local localPoint = water.CFrame:PointToObjectSpace(worldPosition)
	local halfSize = water.Size * 0.5
	local ringRadiusX = ring.Size.X * 0.5
	local ringRadiusZ = ring.Size.Z * 0.5
	local x = math.clamp(localPoint.X, -halfSize.X + ringRadiusX, halfSize.X - ringRadiusX)
	local z = math.clamp(localPoint.Z, -halfSize.Z + ringRadiusZ, halfSize.Z - ringRadiusZ)
	local y = halfSize.Y + (ring.Size.Y * 0.5) + 0.03
	return water.CFrame:PointToWorldSpace(Vector3.new(x, y, z))
end

	local mouse = player:GetMouse()
	local viewport = camera.ViewportSize
	local normalizedX = ((mouse.X / math.max(viewport.X, 1)) - 0.5) * 2
	local normalizedY = ((mouse.Y / math.max(viewport.Y, 1)) - 0.5) * 2
	local base = shared.PondCamera.CFrame

local function getIdleModelCFrame(targetPosition)
	local baseCFrame = shared.ReplaceWithModel.CFrame
	if not targetPosition then
		return baseCFrame
	end

	local flatDirection = Vector3.new(targetPosition.X - baseCFrame.Position.X, 0, targetPosition.Z - baseCFrame.Position.Z)
	if flatDirection.Magnitude < 0.001 then
		return baseCFrame
	end

	local forward = Vector3.new(baseCFrame.LookVector.X, 0, baseCFrame.LookVector.Z)
	if forward.Magnitude < 0.001 then
		forward = Vector3.zAxis
	else
		forward = forward.Unit
	end

	local targetForward = flatDirection.Unit
	local dot = math.clamp(forward:Dot(targetForward), -1, 1)
	local angle = math.acos(dot)
	if forward:Cross(targetForward).Y < 0 then
		angle = -angle
	end

	return CFrame.new(baseCFrame.Position) * CFrame.Angles(0, angle, 0) * (baseCFrame - baseCFrame.Position)
end

local function throwActiveModelAtRing()
	if not state.activeModel or not state.modelPivotValue then
		return
	end

	cancelTween(state.modelTween)
	state.modelTween = nil

	local travelDriver = state.modelPivotValue
	local startCFrame = state.activeModel:GetPivot()
	local targetCFrame = state.lastRingCFrame or shared.DropRing.CFrame
	local targetPosition = targetCFrame.Position + Vector3.new(0, shared.ModelHoverHeight * 0.16, 0)
	local startPosition = startCFrame.Position
	local travelVector = targetPosition - startPosition
	local distance = travelVector.Magnitude
	local baseRotation = startCFrame - startCFrame.Position
	local peakPosition = startPosition:Lerp(targetPosition, 0.5) + Vector3.new(0, math.max(3, distance * 0.12), 0)
	local peakCFrame = CFrame.new(peakPosition) * baseRotation * CFrame.Angles(math.rad(-18), 0, 0)
	local landCFrame = CFrame.new(targetPosition) * baseRotation * CFrame.Angles(math.rad(10), 0, 0)

	travelDriver.Value = startCFrame

	state.modelTween = TweenService:Create(travelDriver, TweenInfo.new(shared.ThrowFlightSeconds * 0.58, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
		Value = peakCFrame,
	})
	state.modelTween:Play()
	state.modelTween.Completed:Wait()

	state.modelTween = TweenService:Create(travelDriver, TweenInfo.new(shared.ThrowFlightSeconds * 0.42, Enum.EasingStyle.Quad, Enum.EasingDirection.In), {
		Value = landCFrame,
	})
	state.modelTween:Play()
	state.modelTween.Completed:Wait()
	state.modelTween = nil

	clearActiveModel()
end

		return

local function rollWeightedFish(targetStrength)
	local totalWeight = 0
	local weighted = {}
	for _, fish in ipairs(config.FishPool) do
		local distance = math.abs((fish.Strength or 1) - targetStrength)
		local weight = math.max(1, 12 - distance)
		weighted[#weighted + 1] = { fish = fish, weight = weight }
		totalWeight += weight
	end

	local pick = math.random() * totalWeight
	local running = 0
	for _, entry in ipairs(weighted) do
		running += entry.weight
		if pick <= running then
			return entry.fish
		end
	end

	return config.FishPool[1]
